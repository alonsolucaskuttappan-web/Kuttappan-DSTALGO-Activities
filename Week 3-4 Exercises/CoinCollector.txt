namespace CoinCollector
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnCollect_Click(object sender, EventArgs e)
        {
            int coins =0;   
            do
            {
                coins += 10;
            }
            while (coins < 100);

            lblCoins.Text = "Coins:"+coins;
        }
    }
}
