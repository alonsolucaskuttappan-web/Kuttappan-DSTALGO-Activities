namespace WinFormsApp14
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            int n = 3;
            int combinations = 1;

            for (int i = 0; i < n; i++)
            {
                combinations = combinations * 2;
            }

            label1.Text = "Number of combinations: " + combinations;
        }
    }
}
