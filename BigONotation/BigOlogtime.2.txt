namespace WinFormsApp9
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            int n = 16;
            string output = "";

            while (n > 1)
            {
                n = n / 2;
                output += n + Environment.NewLine;
            }

            label1.Text = output;
        }
    }
}
