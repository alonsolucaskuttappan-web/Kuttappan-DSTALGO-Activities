Public Class Form1
    Private Sub Button1_Click(sender As Object, e As EventArgs) Handles Button1.Click

        Dim numbers() As Integer = {10, 20, 30, 40, 50}

        Dim output As String = ""

        For i As Integer = 0 To numbers.Length - 1
            output &= numbers(i) & " "
        Next

        Label1.Text = output

    End Sub
End Class
