namespace WinFormsApp16
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            int positions = 4;
            int possibilities = 1;

            for (int i = 0; i < positions; i++)
            {
                possibilities *= 2;
            }

            label1.Text = "Possible passwords: " + possibilities;
        }
    }
}
